# Minimum versions rather than exact pins.
#
# Exact pins are better for reproducibility, and they are what this was
# developed against. They are not used here because Streamlit Community Cloud
# builds on its own Python version, and an exact pin on a recent release is the
# most common reason a deploy fails: the version exists, but there is no wheel
# for the Python that the host happens to be running.
#
# The versions below are the ones this project was built and tested with, used
# as floors. If a future release breaks something, pin that one package back to
# the exact version listed here.

# Data handling
pandas>=2.2          # developed against 3.0.6
numpy>=1.26          # developed against 2.5.3

# Statistics. statsmodels gives regression coefficients with standard errors
# and confidence intervals, which is what the analysis reports.
statsmodels>=0.14    # developed against 0.15.0
scipy>=1.11          # developed against 1.18.1

# Charts written to /outputs
matplotlib>=3.8      # developed against 3.11.2

# Fetching and scraping
requests>=2.31       # developed against 2.34.2
beautifulsoup4>=4.12 # developed against 4.15.0
lxml>=5.0            # developed against 6.1.3
# pypdf reads the district bell schedule documents, which are published as PDFs
pypdf>=4.0           # developed against 6.19.0

# Dashboard
streamlit>=1.35      # developed against 1.64.0

# Config file parsing
PyYAML>=6.0          # developed against 6.0.3
