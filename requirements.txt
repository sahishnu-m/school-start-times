# Runtime dependencies for the pipeline and the dashboard.
#
# Minimum versions rather than exact pins. Streamlit Community Cloud builds on
# its own Python version, and an exact pin on a recent release is a common way
# for a deploy to fail: the version exists, but there is no wheel for the
# Python the host happens to be running.
#
# statsmodels and scipy are deliberately absent. They were used for the
# regressions and the significance tests, and the deployed dashboard failed to
# import them, because both are compiled against a particular numpy and the
# hosted environment installed a combination that did not match. Everything
# they were used for now lives in src/stats_core.py, written against numpy, and
# is checked against both libraries in tests/test_stats_core.py.

# Data handling and all of the statistics
pandas>=2.0
numpy>=1.24

# Charts written to /outputs by the pipeline. The dashboard reads the saved
# images and never imports matplotlib.
matplotlib>=3.7

# Fetching and scraping, used by the pipeline only
requests>=2.31
beautifulsoup4>=4.12
lxml>=5.0
# pypdf reads the district bell schedule documents, which are published as PDFs
pypdf>=4.0

# Dashboard
streamlit>=1.30

# Config file parsing
PyYAML>=6.0
